4.5基于注解开发AspectJ

  • xml

  • ```xml

    <beans xmlns=”http://www.springframework.org/schema/beans"

       xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
   xmlns:aop="http://www.springframework.org/schema/aop"
   xsi:schemaLocation="http://www.springframework.org/schema/beans
   http://www.springframework.org/schema/beans/spring-beans.xsd
    http://www.springframework.org/schema/aop
    http://www.springframework.org/schema/aop/spring-aop.xsd">

<!--定义目标对象-->
<bean id="testDao" class="dynamic.jdk.TestDaoImpl"/>
<!--定义切面-->
<bean id="myAspect" class="xmlAspectJ.MyAspect"/>
<!--aop配置 aop:config标签是开发aspectj的顶层配置元素，在beans下面，可以包含多个-->
<aop:config>
    <!--配置切面-->
    <aop:aspect ref="myAspect">
        <!--配置切入点，通知切面增强哪些方法,相当于配置织入规则-->
        <aop:pointcut id="myPointcut" expression="execution(* dynamic.jdk.TestDaoImpl.*(..))"/>
        <!--将通知与切入点关联-->
        <!--关联前置通知-->
        <aop:before method="before" pointcut-ref="myPointcut"/>
        <!--关联环绕通知-->
        <aop:around method="around" pointcut-ref="myPointcut"/>
        <!--关联异常通知-->
        <aop:after method="after" pointcut-ref="myPointcut"/>
        <!--关联异常通知-->
        <aop:after-throwing method="except" pointcut-ref="myPointcut" throwing="e"/>
        <!--关联后置返回通知-->
        <aop:after-returning method="afterReturning" pointcut-ref="myPointcut"/>
    </aop:aspect>
</aop:config>
